CREATE EXTERNAL TABLE yellow_tripdata_2024 (
  idx INT,  -- corresponds to "Unnamed: 0"
  VendorID INT,
  tpep_pickup_datetime STRING,
  tpep_dropoff_datetime STRING,
  passenger_count INT,
  trip_distance DOUBLE,
  RatecodeID INT,
  store_and_fwd_flag STRING,
  PULocationID INT,
  DOLocationID INT,
  payment_type INT,
  fare_amount DOUBLE,
  extra DOUBLE,
  mta_tax DOUBLE,
  tip_amount DOUBLE,
  tolls_amount DOUBLE,
  improvement_surcharge DOUBLE,
  total_amount DOUBLE,
  congestion_surcharge DOUBLE,
  Airport_fee DOUBLE
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/nyctrips/2024'
TBLPROPERTIES ("skip.header.line.count"="1");



CREATE EXTERNAL TABLE yellow_tripdata_2025 (
  idx INT,  -- corresponds to "Unnamed: 0"
  VendorID INT,
  tpep_pickup_datetime STRING,
  tpep_dropoff_datetime STRING,
  passenger_count INT,
  trip_distance DOUBLE,
  RatecodeID INT,
  store_and_fwd_flag STRING,
  PULocationID INT,
  DOLocationID INT,
  payment_type INT,
  fare_amount DOUBLE,
  extra DOUBLE,
  mta_tax DOUBLE,
  tip_amount DOUBLE,
  tolls_amount DOUBLE,
  improvement_surcharge DOUBLE,
  total_amount DOUBLE,
  congestion_surcharge DOUBLE,
  Airport_fee DOUBLE
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/nyctrips/2025'
TBLPROPERTIES ("skip.header.line.count"="1");



SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date, 
    COUNT(*) AS total_trips,
    '2024' AS year
FROM yellow_tripdata_2024
GROUP BY substr(tpep_pickup_datetime, 1, 10)
UNION ALL
SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date, 
    COUNT(*) AS total_trips,
    '2025' AS year
FROM yellow_tripdata_2025
GROUP BY substr(tpep_pickup_datetime, 1, 10)
ORDER BY pickup_date;



SELECT 
    PULocationID,
    COUNT(*) AS trips_count,
    '2024' AS year
FROM yellow_tripdata_2024
GROUP BY PULocationID
UNION ALL
SELECT 
    PULocationID,
    COUNT(*) AS trips_count,
    '2025' AS year
FROM yellow_tripdata_2025
GROUP BY PULocationID
ORDER BY trips_count DESC;



SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date, 
    AVG(fare_amount) AS avg_fare,
    '2024' AS year
FROM yellow_tripdata_2024
GROUP BY substr(tpep_pickup_datetime, 1, 10)
UNION ALL
SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date, 
    AVG(fare_amount) AS avg_fare,
    '2025' AS year
FROM yellow_tripdata_2025
GROUP BY substr(tpep_pickup_datetime, 1, 10)
ORDER BY pickup_date;






SELECT 
  pickup_date,
  AVG(trip_speed) AS avg_speed,
  year
FROM (
  SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date,
    (trip_distance / ((unix_timestamp(tpep_dropoff_datetime, 'yyyy-MM-dd HH:mm:ss') 
                        - unix_timestamp(tpep_pickup_datetime, 'yyyy-MM-dd HH:mm:ss')) / 3600)) AS trip_speed,
    '2024' AS year
  FROM yellow_tripdata_2024
  WHERE (unix_timestamp(tpep_dropoff_datetime, 'yyyy-MM-dd HH:mm:ss') 
         - unix_timestamp(tpep_pickup_datetime, 'yyyy-MM-dd HH:mm:ss')) > 0
  
  UNION ALL
  
  SELECT 
    substr(tpep_pickup_datetime, 1, 10) AS pickup_date,
    (trip_distance / ((unix_timestamp(tpep_dropoff_datetime, 'yyyy-MM-dd HH:mm:ss') 
                        - unix_timestamp(tpep_pickup_datetime, 'yyyy-MM-dd HH:mm:ss')) / 3600)) AS trip_speed,
    '2025' AS year
  FROM yellow_tripdata_2025
  WHERE (unix_timestamp(tpep_dropoff_datetime, 'yyyy-MM-dd HH:mm:ss') 
         - unix_timestamp(tpep_pickup_datetime, 'yyyy-MM-dd HH:mm:ss')) > 0
) a
GROUP BY pickup_date, year
ORDER BY pickup_date;
